# Manual test runbook
#
# The automated suite (npm test) covers the same ground and more; this file is
# for poking at a running server by hand, and for the multi-server checks that
# need several terminals.
#
# Every URL uses 127.0.0.1 rather than localhost: on some systems localhost
# resolves to ::1 first and the server may only be bound to IPv4.

# ============================================================================
# PART 1: PREREQUISITES
# ============================================================================

node --version          # expected: v18.17 or newer
redis-cli --version
redis-cli ping          # expected: PONG
npm ci                  # expected: clean install, no "invalid" entries
npm ls --depth=0        # expected: express + redis only
npm audit               # expected: found 0 vulnerabilities
npm test                # expected: 45 tests, 0 fail (Redis tests skip without Redis)

# ============================================================================
# PART 2: IN-MEMORY SERVER (single process)
# ============================================================================

# 2.1 - start it
npm start
# expected: "[Server] RateGuard demo (in-memory) listening on http://localhost:3000"
#           plus a note that ADMIN_TOKEN is unset

# 2.2 - connectivity
curl -s http://127.0.0.1:3000/health
# expected: {"status":"ok","backend":"memory",...}

# 2.3 - strict endpoint, 5 req/min
for i in $(seq 1 7); do curl -s -o /dev/null -w '%{http_code} ' http://127.0.0.1:3000/api/expensive; done
# expected: 200 200 200 200 200 429 429

# 2.4 - the relaxed endpoint is unaffected by the strict one being exhausted
curl -s -o /dev/null -w '%{http_code}\n' http://127.0.0.1:3000/api/public
# expected: 200   (regression: a single shared bucket used to return 429 here)

# 2.5 - headers
curl -si http://127.0.0.1:3000/api/expensive | grep -iE 'ratelimit|retry-after'
# expected: RateLimit-Limit: 5, RateLimit-Remaining: N, RateLimit-Reset: <seconds>,
#           X-RateLimit-Reset: <ISO timestamp>, and Retry-After only on a 429 (never 0)

# 2.6 - a recognised API key gets its own bucket
curl -s -H 'X-Api-Key: demo-key-1' http://127.0.0.1:3000/api/with-key
# expected: {"recognisedKey":true,...}

# 2.7 - rotating unrecognised keys must NOT mint new quota
for i in $(seq 1 14); do curl -s -o /dev/null -w '%{http_code} ' -H "X-Api-Key: rot$i" http://127.0.0.1:3000/api/with-key; done
# expected: ten 200s then 429s - unknown keys fall back to the IP bucket

# 2.8 - login protection, custom 429 body
for i in $(seq 1 4); do curl -s -o /dev/null -w '%{http_code} ' -X POST http://127.0.0.1:3000/api/login; done
curl -s -X POST http://127.0.0.1:3000/api/login
# expected: 200 200 200 429, then {"error":"RATE_LIMIT_EXCEEDED",...}

# 2.9 - refill: exhaust, wait the advertised Retry-After, retry once
for i in $(seq 1 6); do curl -s -o /dev/null http://127.0.0.1:3000/api/expensive; done
RETRY=$(curl -si http://127.0.0.1:3000/api/expensive | awk '/[Rr]etry-[Aa]fter/ {print $2}' | tr -d '\r')
sleep "$RETRY"
curl -s -o /dev/null -w '%{http_code}\n' http://127.0.0.1:3000/api/expensive
# expected: 200 - waiting exactly Retry-After must be enough

# 2.10 - admin routes are absent unless ADMIN_TOKEN is set
curl -s -o /dev/null -w '%{http_code}\n' http://127.0.0.1:3000/api/admin/stats
# expected: 404

# 2.11 - graceful shutdown
#   press Ctrl+C, or: kill -TERM <node pid>
# expected: "SIGTERM received, draining connections..." then "shutdown complete", exit 0

# ============================================================================
# PART 3: REDIS SERVER
# ============================================================================

# 3.1 - fresh start
redis-cli FLUSHALL
export ADMIN_TOKEN=$(openssl rand -hex 24)
npm run start:redis
# expected: "[RedisStore] connection ready", "[Server] Redis connected"

# 3.2 - health reflects the dependency
curl -s http://127.0.0.1:3000/health
# expected: {"status":"ok","redis":{"healthy":true,"latencyMs":N},"failOpen":true,...}

# 3.3 - strict endpoint over Redis
for i in $(seq 1 7); do curl -s -o /dev/null -w '%{http_code} ' http://127.0.0.1:3000/api/expensive; done
# expected: 200 x5 then 429 x2

# 3.4 - keys are namespaced per limiter, not per client
redis-cli KEYS 'ratelimit:*'
# expected: ratelimit:global|<ip>, ratelimit:expensive|<ip>, ...

# 3.5 - TTL is derived from the refill time, not fixed at an hour
redis-cli TTL "ratelimit:expensive|127.0.0.1"
# expected: <= 120   (5 tokens at 5/min = 60s to refill, + 60s padding)

# 3.6 - admin endpoints require the bearer token
curl -s -o /dev/null -w '%{http_code}\n' http://127.0.0.1:3000/api/admin/stats
# expected: 401
curl -s -H "Authorization: Bearer $ADMIN_TOKEN" http://127.0.0.1:3000/api/admin/stats
# expected: counts and byScope only - no identifiers in the response

# 3.7 - stats must not leak an API key
curl -s -H 'X-Api-Key: SECRET-KEY-DO-NOT-LEAK' http://127.0.0.1:3000/api/with-key > /dev/null
curl -s -H "Authorization: Bearer $ADMIN_TOKEN" http://127.0.0.1:3000/api/admin/stats | grep -c SECRET-KEY
# expected: 0

# 3.8 - reset one scope
curl -s -X POST -H "Authorization: Bearer $ADMIN_TOKEN" \
  'http://127.0.0.1:3000/api/admin/reset/127.0.0.1?scope=expensive'
curl -s -o /dev/null -w '%{http_code}\n' http://127.0.0.1:3000/api/expensive
# expected: {"removed":1,...} then 200

# 3.9 - inspect a bucket
curl -s -H "Authorization: Bearer $ADMIN_TOKEN" \
  'http://127.0.0.1:3000/api/admin/bucket/127.0.0.1?scope=expensive'
# expected: {"state":{"tokens":N,"lastRefill":<ms>,"ttlSeconds":N,"key":"..."}}

# ============================================================================
# PART 4: DISTRIBUTED (two servers, one Redis)
# ============================================================================

# 4.1 - terminal 1
redis-cli FLUSHALL
redis-server

# 4.2 - terminal 2
PORT=3000 SERVER_ID=server1 npm run start:redis

# 4.3 - terminal 3
PORT=3001 SERVER_ID=server2 npm run start:redis

# 4.4 - terminal 4: alternate between them, 5 req/min total
for i in $(seq 1 4); do
  printf '3000:%s 3001:%s\n' \
    "$(curl -s -o /dev/null -w '%{http_code}' http://127.0.0.1:3000/api/expensive)" \
    "$(curl -s -o /dev/null -w '%{http_code}' http://127.0.0.1:3001/api/expensive)"
done
# expected: five 200s in total across both servers, then 429 from both

# 4.5 - both servers report the same bucket
curl -s -H "Authorization: Bearer $ADMIN_TOKEN" \
  'http://127.0.0.1:3001/api/admin/bucket/127.0.0.1?scope=expensive'
# expected: the same tokens/last_refill the other server would report

# ============================================================================
# PART 5: FAILURE MODES
# ============================================================================

# 5.1 - Redis down: requests must be answered, fast
redis-cli SHUTDOWN NOSAVE
for i in $(seq 1 8); do curl -s -o /dev/null -m 10 -w '%{http_code} in %{time_total}s\n' http://127.0.0.1:3000/api/expensive; done
# expected: 200 in ~0.00Xs each (failOpen default). Never a hang, never a 000.
# expected: the log shows a handful of throttled errors, not one per request

# 5.2 - health reports the outage
curl -s -o /dev/null -w '%{http_code}\n' http://127.0.0.1:3000/health
# expected: 503

# 5.3 - fail closed instead
#   restart with: FAIL_OPEN=0 npm run start:redis, then kill Redis and repeat 5.1
# expected: 429 with Retry-After: 1

# 5.4 - recovery without restarting the app
redis-server &
sleep 2
for i in $(seq 1 7); do curl -s -o /dev/null -w '%{http_code} ' http://127.0.0.1:3000/api/expensive; done
# expected: enforcement resumes (200 x5 then 429); the script is reloaded automatically

# 5.5 - refuse to start without Redis, if that is what you want
REQUIRE_REDIS=1 REDIS_PORT=1 npm run start:redis
# expected: a clear error and exit 1

# ============================================================================
# PART 6: LOAD
# ============================================================================

npm run bench
BENCH_REQUESTS=20000 BENCH_CONCURRENCY=100 npm run bench
# reports rps and p50/p95/p99 for no-limiter, MemoryStore and RedisStore.
